async function getFingerprint(){
  try {
    var data = [navigator.userAgent, navigator.language, screen.width + 'x' + screen.height, screen.colorDepth, new Date().getTimezoneOffset()].join('|');
    var buf = await crypto.subtle.digest('SHA-256', new TextEncoder().encode(data));
    return Array.from(new Uint8Array(buf)).slice(0,16).map(function(b){ return b.toString(16).padStart(2,'0'); }).join('');
  } catch(e){ return getDeviceId(); }
}

function escapeHtml(s){
  return String(s == null ? '' : s).replace(/[&<>"']/g, function(m){
    return {'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[m];
  });
}

function fmtTime(sec){
  sec = Math.floor(sec || 0);
  var h = Math.floor(sec/3600), m = Math.floor((sec%3600)/60), s = sec%60;
  if (h > 0) return h + 'h ' + m + 'm';
  if (m > 0) return m + 'm ' + s + 's';
  return s + 's';
}
function fmtClock(sec){
  sec = Math.floor(sec || 0);
  var h = Math.floor(sec/3600), m = Math.floor((sec%3600)/60), s = sec%60;
  if (h > 0) return h + ':' + (m<10?'0':'') + m + ':' + (s<10?'0':'') + s;
  return m + ':' + (s<10?'0':'') + s;
}
function timeAgo(ts){
  var d = new Date(ts).getTime(); var diff = Date.now() - d;
  if (diff < 60000) return 'just now';
  if (diff < 3600000) return Math.floor(diff/60000) + 'm ago';
  if (diff < 86400000) return Math.floor(diff/3600000) + 'h ago';
  return Math.floor(diff/86400000) + 'd ago';
}
function debounce(fn, ms){
  var t;
  return function(){ var a = arguments, self = this; clearTimeout(t); t = setTimeout(function(){ fn.apply(self, a); }, ms); };
}

function startMaintenanceParticles(){
  var canvas = $('#maintenance-particles');
  if (!canvas) return;
  var ctx = canvas.getContext('2d');
  var W, H, particles = [];
  function resize(){ W = canvas.width = window.innerWidth; H = canvas.height = window.innerHeight; }
  window.addEventListener('resize', resize);
  resize();
  var COUNT = 80;
  function Particle(){
    this.x = Math.random() * W;
    this.y = Math.random() * H;
    this.size = Math.random() * 2.5 + 0.5;
    this.speedX = (Math.random() - 0.5) * 0.3;
    this.speedY = (Math.random() - 0.5) * 0.3;
    this.opacity = Math.random() * 0.5 + 0.1;
  }
  Particle.prototype.update = function(){
    this.x += this.speedX; this.y += this.speedY;
    if (this.x < 0 || this.x > W) this.speedX *= -1;
    if (this.y < 0 || this.y > H) this.speedY *= -1;
  };
  Particle.prototype.draw = function(){
    ctx.beginPath();
    ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2);
    ctx.fillStyle = 'rgba(255,170,68,' + this.opacity + ')';
    ctx.fill();
  };
  for (var i = 0; i < COUNT; i++) particles.push(new Particle());
  function drawLines(){
    for (var i = 0; i < particles.length; i++){
      for (var j = i + 1; j < particles.length; j++){
        var dx = particles[i].x - particles[j].x, dy = particles[i].y - particles[j].y;
        var dist = Math.sqrt(dx*dx + dy*dy);
        if (dist < 150){
          ctx.beginPath();
          ctx.moveTo(particles[i].x, particles[i].y);
          ctx.lineTo(particles[j].x, particles[j].y);
          ctx.strokeStyle = 'rgba(255,170,68,' + (0.06 * (1 - dist/150)) + ')';
          ctx.lineWidth = 0.6; ctx.stroke();
        }
      }
    }
  }
  function animate(){
    if (maintenanceParticleRAF) cancelAnimationFrame(maintenanceParticleRAF);
    ctx.clearRect(0, 0, W, H);
    for (var i = 0; i < particles.length; i++){ particles[i].update(); particles[i].draw(); }
    drawLines();
    maintenanceParticleRAF = requestAnimationFrame(animate);
  }
  animate();
}

// ============================================================
//  DEVTOOLS & IDM DETECTION (Debugger Trap + Logout)
// ============================================================
function startDevToolsDetection(){
  try { if (window.self !== window.top) return; } catch(e){ return; }
  
  var securityTriggered = false;

  function handleThreat(reason) {
    if (securityTriggered) return;
    securityTriggered = true;
    logDevTools(reason);
    toast('Security Alert: ' + reason + '. Logging out...', 'error');
    setTimeout(function(){ 
      localStorage.removeItem('vibhuux_session'); 
      location.reload(); 
    }, 1500);
  }

  // 1. Keyboard Shortcuts Block
  document.addEventListener('keydown', function(e){
    var key = (e.key || '').toUpperCase();
    if (e.key === 'F12' || (e.ctrlKey && e.shiftKey && (key === 'I' || key === 'J' || key === 'C')) || (e.ctrlKey && key === 'U')){
      e.preventDefault();
      handleThreat('Restricted Shortcut');
      return false;
    }
  });

  // 2. IDM Detection & Debugger Trap (Only runs if not embedded)
  setInterval(function(){
    if (securityTriggered) return;
    
    // Check for IDM injection elements
    var idmElements = document.querySelectorAll('[id*="idm" i], [class*="idm" i]');
    if (idmElements.length > 0 && !document.getElementById('video-stage').contains(idmElements[0])) {
       handleThreat('IDM Downloader detected');
       return;
    }

    // Debugger Trap (Time Difference)
    var startTime = performance.now();
    debugger; // Browser pauses here if DevTools is open
    var endTime = performance.now();
    
    // If execution took > 100ms, DevTools is open
    if (endTime - startTime > 100) {
      handleThreat('Developer Tools detected');
    }
  }, 2000);
}

async function logDevTools(reason){
  try {
    await fetch('/api/security/devtools-detected', {
      method: 'POST', headers: {'Content-Type':'application/json'},
      body: JSON.stringify({
        userId: STATE.user ? STATE.user.id : null,
        deviceId: STATE.deviceId,
        telegram_id: STATE.profile ? STATE.profile.telegram_id : null,
        telegram_username: STATE.profile ? STATE.profile.telegram_username : null,
        telegram_name: STATE.profile ? STATE.profile.telegram_name : null,
        email: STATE.user ? STATE.user.email : null,
        website_username: STATE.profile ? STATE.profile.username : null,
        reason: reason || 'Unknown'
      })
    });
  } catch(e){}
}

async function refreshSession() {
  if (!STATE.session || !STATE.session.refresh_token) return false;
  try {
    var r = await fetch('/api/auth/refresh', {
      method: 'POST', headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({ refresh_token: STATE.session.refresh_token })
    });
    if (!r.ok) return false;
    var data = await r.json();
    if (data.session && data.session.access_token) {
      STATE.session = data.session;
      STATE.user = data.session.user;
      localStorage.setItem('vibhuux_session', JSON.stringify(data.session));
      return true;
    }
    return false;
  } catch(e) { return false; }
}

async function loadPaidBatches(){

    var r = await fetch('/api/user/paid-batches', { headers: {'Authorization': 'Bearer ' + STATE.session.access_token} });
    
    // 🆕 Agar 401 aaye (Token expire), toh token refresh karke dobara try karo
    if (r.status === 401) {
      var refreshed = await refreshSession();
      if (refreshed) {
        r = await fetch('/api/user/paid-batches', { headers: {'Authorization': 'Bearer ' + STATE.session.access_token} });
      }
    }
    
    var data = await r.json();
    STATE.userPaidBatches = new Set(data.batch_ids || []);
    
    // 🆕 Role ko sync karo
    if (STATE.profile) {
      var isPaid = (STATE.profile.role === 'paid') || (STATE.userPaidBatches.size > 0);
      STATE.profile.role = isPaid ? 'paid' : 'free';
    }
    updateSidebar();
  } catch(e){ STATE.userPaidBatches = new Set(); }
}

        headers: {'Authorization': 'Bearer ' + STATE.session.access_token}
      }).then(function(x){ return x.json(); }),
      fetch('/api/subject/progress?subject_id=' + encodeURIComponent(subjectId), {
        headers: {'Authorization': 'Bearer ' + STATE.session.access_token}
      }).then(function(x){ return x.json(); }).catch(function(){ return { progress: [] }; })
    ]);
    var lectures = results[0].lectures || [];
    var progressList = results[1].progress || [];
    var progMap = {};
    progressList.forEach(function(p){ progMap[p.lecture_id] = p; });
    var hasBatchAccess = batchId && STATE.userPaidBatches.has(batchId);

    var html = backButton() + '<h2 class="list-title"><i class="fas fa-' + (type === 'video' ? 'video' : 'file-pdf') + '" style="color:' + (type === 'video' ? '#ef4444' : '#3b82f6') + '"></i> ' + (type === 'video' ? 'Video Lectures' : 'PDF Notes') + '</h2>';
    if (hasBatchAccess){
      html += '<div style="background:rgba(34,197,94,.1);border:1px solid rgba(34,197,94,.3);border-radius:14px;padding:12px 16px;margin-bottom:16px;font-size:.85rem;color:#4ade80;"><i class="fas fa-crown"></i> <strong>Premium Access</strong> — Iss batch ke saare lectures aapko turant available hain.</div>';
    }

    if (!lectures.length){
      html += '<div class="empty-state"><i class="fas fa-inbox"></i><h3>No content</h3></div>';
    } else {
      html += '<div>';
      lectures.forEach(function(l){
        var isLocked = l._locked;
        var isPaid = l._paid;
        var releaseRaw = l.release_date || l.release_at || l.unlock_date || l.unlock_at;
        var releaseInfo = '';
        if (releaseRaw){
          try { releaseInfo = new Date(releaseRaw).toLocaleDateString('en-IN', { day:'2-digit', month:'short', year:'numeric' }); } catch(e){}
        }
        var lectureTitle = escapeHtml(l.topic || l.file_name || 'Untitled');
        var lectureId = escapeJs(l.id);
        var lectureType = type;
        var prog = progMap[l.id] || {};
        var watchedSec = prog.seconds_watched || 0;
        var completed = prog.completed === true;
        var actionsHtml = '';

        if (isLocked){
          actionsHtml = '<button class="btn-locked"><i class="fas fa-lock"></i> Locked</button>';
        } else if (completed){
          actionsHtml = '<button class="btn-watch" onclick="window._openLecture(\'' + lectureId + '\',\'' + escapeJs(l.topic || l.file_name) + '\',\'' + lectureType + '\')"><i class="fas fa-redo"></i> Rewatch</button>';
        } else if (watchedSec > 0){
          actionsHtml = '<button class="btn-continue" onclick="window._openLecture(\'' + lectureId + '\',\'' + escapeJs(l.topic || l.file_name) + '\',\'' + lectureType + '\')"><i class="fas fa-play"></i> Continue</button>';
        } else {
          actionsHtml = '<button class="btn-watch" onclick="window._openLecture(\'' + lectureId + '\',\'' + escapeJs(l.topic || l.file_name) + '\',\'' + lectureType + '\')"><i class="fas fa-play"></i> Watch</button>';
        }

        var progressHtml = '';
        if (!isLocked && (watchedSec > 0 || completed)){
          var pct = completed ? 100 : Math.min(95, Math.round((watchedSec/Math.max(1, 600))*100));
          progressHtml = '<div class="lecture-progress-mini"><div class="lecture-progress-mini-fill ' + (completed ? 'done' : '') + '" style="width:' + pct + '%"></div></div>';
        }

        var unlockHtml = '';
        if (isLocked && releaseInfo && !isPaid){
          unlockHtml = '<div class="lecture-unlock-tag"><i class="fas fa-calendar-alt"></i> Unlock on: ' + releaseInfo + '</div>';
        } else if (isLocked && isPaid){
          unlockHtml = '<div class="lecture-unlock-tag"><i class="fas fa-crown"></i> Premium — paid access chahiye</div>';
        }

        html += '<div class="lecture-item ' + (isLocked ? 'locked' : '') + ' ' + (completed ? 'watched' : '') + '"><div class="lecture-icon">' + (isLocked ? '🔒' : (type === 'video' ? '🎬' : '📄')) + (completed ? '<div class="lecture-watched-badge"><i class="fas fa-check"></i></div>' : '') + '</div><div class="lecture-info"><div class="lecture-title">' + lectureTitle + '</div><div class="lecture-meta">' + (l.file_name ? '<span>' + escapeHtml(l.file_name) + '</span>' : '') + (watchedSec > 0 && !completed ? '<span style="color:#93c5fd;">' + fmtTime(watchedSec) + '</span>' : '') + (completed ? '<span style="color:#4ade80;"><i class="fas fa-check"></i> Completed</span>' : '') + '</div>' + unlockHtml + progressHtml + '</div><div class="lecture-actions">' + actionsHtml + '</div></div>';
      });
      html += '</div>';
    }
    v.innerHTML = html;
  } catch(e){
    v.innerHTML = '<div class="empty-state"><p>' + escapeHtml(e.message) + '</p></div>';
  }
};

  return openPDFViewer(id, title);
};

async function openVideoPlayer(id, title){
  resetPlayerState();
  $('#video-stage-title').textContent = title || 'Loading...';
  $('#vp-thumb').classList.remove('hide');
  $('#vp-crop').innerHTML = '';
  $('#vp-box').className = 'vp-box';
  $('#vp-center-pause').classList.remove('show');
  $('#vp-progress-fill').style.width = '0%';
  $('#vp-time-current').textContent = '0:00';
  $('#vp-time-total').textContent = '0:00';
  var dlBtn = $('#vc-download');
  dlBtn.disabled = false;
  dlBtn.querySelector('span').textContent = 'Download';
  $('#video-stage').style.display = 'flex';
  document.body.style.overflow = 'hidden';

  try {
    var r = await fetch('/api/lecture/detail?id=' + id, { headers: {'Authorization': 'Bearer ' + STATE.session.access_token} });
    var data = await r.json();
    if (!data.lecture) throw new Error('Lecture not found');
    PLAYER.lecture = data.lecture;
    var url = data.lecture.video_url || '';
    var ytId = getYouTubeId(url);
    var driveId = getDriveFileId(url);
    var isStream = isStreamingWorker(url);
    var isHls = isHlsUrl(url);

    if (isStream){ PLAYER.mode = 'streaming'; $('#vp-box').classList.add('mode-streaming'); }
    else if (ytId){ PLAYER.mode = 'youtube'; PLAYER.ytId = ytId; $('#vp-box').classList.add('mode-youtube'); }
    else if (driveId){ PLAYER.mode = 'drive'; PLAYER.driveId = driveId; $('#vp-box').classList.add('mode-drive'); }
    else if (isHls){ PLAYER.mode = 'hls'; }
    else { PLAYER.mode = 'mp4'; }

    try {
      var pr = await fetch('/api/lecture/progress?lecture_id=' + id, { headers: {'Authorization': 'Bearer ' + STATE.session.access_token} });
      var pd = await pr.json();
      if (pd.progress && pd.progress.seconds_watched > 0){
        PLAYER.watchSeconds = pd.progress.seconds_watched;
        PLAYER.lastSentSec = pd.progress.seconds_watched;
        PLAYER.lastPosition = pd.progress.last_position || 0;
      }
    } catch(e){}

    bindThumbEvents();
  } catch(e){
    toast(e.message, 'error');
    window._closeVideoStage();
  }
}

function bindThumbEvents(){
  var thumb = $('#vp-thumb');
  var playBtn = $('#vp-play-btn');
  var newPlay = playBtn.cloneNode(true);
  playBtn.parentNode.replaceChild(newPlay, playBtn);
  newPlay.addEventListener('click', function(e){
    e.stopPropagation();
    startPlayerAndFullscreen();
  });
  thumb.onclick = function(e){
    if (e.target === thumb || e.target.classList.contains('vp-brand') || e.target.classList.contains('vp-brand-1') || e.target.classList.contains('vp-brand-2') || e.target.classList.contains('vp-brand-3')){
      startPlayerAndFullscreen();
    }
  };
}

function startPlayerAndFullscreen(){
  startPlayer();
  setTimeout(function(){
    try {
      var box = $('#vp-box');
      if (box.requestFullscreen) box.requestFullscreen().catch(function(){});
      else if (box.webkitRequestFullscreen) box.webkitRequestFullscreen();
    } catch(e){}
  }, 300);
}

function startPlayer(){
  if (PLAYER.hasStarted) return;
  PLAYER.hasStarted = true;
  $('#vp-thumb').classList.add('hide');
  var url = PLAYER.lecture.video_url;

  if (PLAYER.mode === 'streaming'){
    var iframe = document.createElement('iframe');
    iframe.src = url + (url.indexOf('?') >= 0 ? '&' : '?') + 'embed=1';
    iframe.allow = 'autoplay; encrypted-media; picture-in-picture; fullscreen';
    iframe.allowFullscreen = true;
    iframe.referrerPolicy = 'no-referrer-when-downgrade';
    iframe.setAttribute('scrolling', 'no');
    $('#vp-crop').appendChild(iframe);
    PLAYER.iframeEl = iframe;
    PLAYER.isReady = true;
    startPlayerTick();
    startWatchTracking();
    setTimeout(function(){ try { iframe.contentWindow.postMessage({ type:'atp-cmd', cmd:'start' }, '*'); } catch(e){} }, 600);
  } else if (PLAYER.mode === 'youtube'){
    loadYTAPI(function(){
      var container = document.createElement('div');
      container.id = 'yt-container-' + Date.now();
      container.style.cssText = 'position:absolute;top:0;left:0;width:100%;height:100%';
      $('#vp-crop').appendChild(container);
      PLAYER.ytPlayer = new YT.Player(container, {
        videoId: PLAYER.ytId, width: '100%', height: '100%',
        playerVars: { autoplay:1, controls:0, modestbranding:1, rel:0, iv_load_policy:3, disablekb:1, fs:0, playsinline:1, cc_load_policy:0, origin: location.origin },
        events: { onReady: onYTReady, onStateChange: onYTState, onError: onYTError }
      });
    });
  } else if (PLAYER.mode === 'drive'){
    var iframe2 = document.createElement('iframe');
    iframe2.src = 'https://drive.google.com/file/d/' + PLAYER.driveId + '/preview';
    iframe2.allow = 'autoplay; encrypted-media';
    iframe2.allowFullscreen = true;
    $('#vp-crop').appendChild(iframe2);
    PLAYER.iframeEl = iframe2;
    PLAYER.isReady = true;
    startPlayerTick();
    startWatchTracking();
  } else {
    var video = document.createElement('video');
    video.controls = false;
    video.playsInline = true;
    video.autoplay = true;
    video.preload = 'auto';
    video.controlsList = 'nodownload';
    if (PLAYER.mode === 'hls'){
      if (video.canPlayType('application/vnd.apple.mpegurl')) video.src = url;
      else if (window.Hls && Hls.isSupported()){ PLAYER.hls = new Hls(); PLAYER.hls.loadSource(url); PLAYER.hls.attachMedia(video); }
      else video.src = url;
    } else {
      video.src = url;
    }
    video.addEventListener('loadedmetadata', function(){
      PLAYER.duration = video.duration || 0;
      if (PLAYER.lastPosition > 0 && PLAYER.lastPosition < video.duration - 5) {
        try { video.currentTime = PLAYER.lastPosition; } catch(e){}
      }
      updateTimeUI();
    });
    video.addEventListener('timeupdate', function(){
      PLAYER.currentTime = video.currentTime;
      PLAYER.lastPosition = video.currentTime;
      updateTimeUI();
    });
    video.addEventListener('play', function(){ PLAYER.isPlaying = true; updatePlayPauseUI(); });
    video.addEventListener('pause', function(){ PLAYER.isPlaying = false; updatePlayPauseUI(); });
    video.addEventListener('contextmenu', function(e){ e.preventDefault(); });
    $('#vp-crop').appendChild(video);
    PLAYER.videoEl = video;
    PLAYER.isReady = true;
    startPlayerTick();
    startWatchTracking();
  }
}

function loadYTAPI(cb){
  if (window.YT && window.YT.Player) return cb();
  window.__ytApiCallbacks = window.__ytApiCallbacks || [];
  window.__ytApiCallbacks.push(cb);
  if (!window.__ytApiLoading){
    window.__ytApiLoading = true;
    window.onYouTubeIframeAPIReady = function(){
      var cbs = window.__ytApiCallbacks || [];
      window.__ytApiCallbacks = [];
      cbs.forEach(function(f){ try { f(); } catch(e){} });
    };
    var tag = document.createElement('script');
    tag.src = 'https://www.youtube.com/iframe_api';
    document.head.appendChild(tag);
  }
}

function onYTReady(e){
  PLAYER.isReady = true;
  try {
    e.target.playVideo(); e.target.unMute(); e.target.setVolume(100);
    if (PLAYER.lastPosition > 0) e.target.seekTo(PLAYER.lastPosition, true);
  } catch(err){}
  startPlayerTick();
  startWatchTracking();
}
function onYTState(e){
  if (e.data === 1){ PLAYER.isPlaying = true; updatePlayPauseUI(); }
  else if (e.data === 2){ PLAYER.isPlaying = false; updatePlayPauseUI(); }
  else if (e.data === 0){ PLAYER.isPlaying = false; updatePlayPauseUI(); }
}
function onYTError(e){
  var codes = { 2:'Invalid video ID', 5:'HTML5 player error', 100:'Video not found / private', 101:'Embed disabled', 150:'Embed disabled' };
  toast('YouTube Error: ' + (codes[e.data] || ('Code ' + e.data)), 'error');
}

function startPlayerTick(){
  if (PLAYER.tickTimer) clearInterval(PLAYER.tickTimer);
  PLAYER.tickTimer = setInterval(function(){
    if (PLAYER.mode === 'youtube' && PLAYER.ytPlayer && PLAYER.ytPlayer.getCurrentTime){
      try {
        PLAYER.currentTime = PLAYER.ytPlayer.getCurrentTime() || 0;
        PLAYER.duration = PLAYER.ytPlayer.getDuration() || 0;
        PLAYER.lastPosition = PLAYER.currentTime;
        updateTimeUI();
      } catch(e){}
    } else if (PLAYER.videoEl){
      PLAYER.currentTime = PLAYER.videoEl.currentTime || 0;
      PLAYER.duration = PLAYER.videoEl.duration || 0;
      updateTimeUI();
    }
  }, 500);
}

function updateTimeUI(){
  var cur = $('#vp-time-current'), tot = $('#vp-time-total'), fill = $('#vp-progress-fill');
  if (cur) cur.textContent = fmtClock(PLAYER.currentTime);
  if (tot) tot.textContent = fmtClock(PLAYER.duration);
  if (fill && PLAYER.duration > 0){
    var pct = Math.min(100, Math.round((PLAYER.currentTime / PLAYER.duration) * 100));
    fill.style.width = pct + '%';
  }
}
function updatePlayPauseUI(){
  var btn = $('#vc-play-pause');
  if (!btn) return;
  btn.innerHTML = PLAYER.isPlaying ? '<i class="fas fa-pause"></i>' : '<i class="fas fa-play"></i>';
}

function showCenterPauseBtn(){
  var c = $('#vp-center-pause');
  if (!c) return;
  c.textContent = PLAYER.isPlaying ? '❚❚' : '▶';
  c.classList.add('show');
  clearTimeout(PLAYER.centerBtnTimer);
  PLAYER.centerBtnTimer = setTimeout(function(){ c.classList.remove('show'); }, 1500);
}

function seekBy(delta){
  if (PLAYER.mode === 'youtube' && PLAYER.ytPlayer && PLAYER.ytPlayer.seekTo){
    try { var t = (PLAYER.ytPlayer.getCurrentTime() || 0) + delta; PLAYER.ytPlayer.seekTo(Math.max(0, t), true); } catch(e){}
  } else if (PLAYER.videoEl){
    try { PLAYER.videoEl.currentTime = Math.max(0, PLAYER.videoEl.currentTime + delta); } catch(e){}
  } else if (PLAYER.mode === 'streaming' && PLAYER.iframeEl){
    try { PLAYER.iframeEl.contentWindow.postMessage({ type:'atp-cmd', cmd: delta > 0 ? 'seekFwd' : 'seekBack' }, '*'); } catch(e){}
  }
  var ind = delta > 0 ? $('#vp-skip-right') : $('#vp-skip-left');
  if (ind){ ind.classList.add('show'); setTimeout(function(){ ind.classList.remove('show'); }, 700); }
}

function seekTo(target){
  if (target < 0) target = 0;
  if (PLAYER.mode === 'youtube' && PLAYER.ytPlayer){
    try { PLAYER.ytPlayer.seekTo(target, true); } catch(e){}
  } else if (PLAYER.videoEl){
    try { PLAYER.videoEl.currentTime = target; } catch(e){}
  } else if (PLAYER.mode === 'streaming' && PLAYER.iframeEl){
    try { PLAYER.iframeEl.contentWindow.postMessage({ type:'atp-cmd', cmd:'seek', args:[target] }, '*'); } catch(e){}
  }
  PLAYER.currentTime = target;
  updateTimeUI();
}

function togglePlayPause(){
  if (PLAYER.mode === 'youtube' && PLAYER.ytPlayer){
    try { if (PLAYER.isPlaying) PLAYER.ytPlayer.pauseVideo(); else PLAYER.ytPlayer.playVideo(); } catch(e){}
  } else if (PLAYER.videoEl){
    try { if (PLAYER.videoEl.paused) PLAYER.videoEl.play(); else PLAYER.videoEl.pause(); } catch(e){}
  } else if (PLAYER.mode === 'streaming' && PLAYER.iframeEl){
    try { PLAYER.iframeEl.contentWindow.postMessage({ type:'atp-cmd', cmd:'toggle' }, '*'); } catch(e){}
  }
  showCenterPauseBtn();
}

function setSpeed(v){
  if (PLAYER.mode === 'youtube' && PLAYER.ytPlayer){ try { PLAYER.ytPlayer.setPlaybackRate(v); } catch(e){} }
  else if (PLAYER.videoEl){ try { PLAYER.videoEl.playbackRate = v; } catch(e){} }
  else if (PLAYER.mode === 'streaming' && PLAYER.iframeEl){
    try { PLAYER.iframeEl.contentWindow.postMessage({ type:'atp-cmd', cmd:'speed', args:[v] }, '*'); } catch(e){}
  }
}

function toggleFullscreen(){
  var box = $('#vp-box'); if (!box) return;
  var fsEl = document.fullscreenElement || document.webkitFullscreenElement || document.mozFullScreenElement || document.msFullscreenElement;
  try {
    if (fsEl){
      if (document.exitFullscreen) document.exitFullscreen();
      else if (document.webkitExitFullscreen) document.webkitExitFullscreen();
    } else {
      if (box.requestFullscreen) box.requestFullscreen();
      else if (box.webkitRequestFullscreen) box.webkitRequestFullscreen();
      else if (box.mozRequestFullScreen) box.mozRequestFullScreen();
      else {
        var v = $('#vp-crop video');
        if (v && v.webkitEnterFullscreen) v.webkitEnterFullscreen();
        else toast('Fullscreen not supported', 'error');
      }
    }
  } catch(err){ toast('Fullscreen fail: ' + err.message, 'error'); }
}

function handleMouseActivity(){
  if (!PLAYER.isFullscreen) return;
  document.body.classList.remove('vs-hidden-ui');
  clearTimeout(PLAYER.fsHideTimer);
  PLAYER.fsHideTimer = setTimeout(function(){
    if (PLAYER.isFullscreen && PLAYER.isPlaying){
      document.body.classList.add('vs-hidden-ui');
    }
  }, 3000);
}

function bindFullscreenEvents(){
  if (window._atpFsBound) return;
  window._atpFsBound = true;
  var onFs = function(){
    var fsEl = document.fullscreenElement || document.webkitFullscreenElement || document.mozFullScreenElement || document.msFullscreenElement;
    var btn = $('#vc-fullscreen');
    if (btn) btn.innerHTML = fsEl ? '<i class="fas fa-compress"></i>' : '<i class="fas fa-expand"></i>';
    PLAYER.isFullscreen = !!fsEl;
    if (fsEl){
      document.body.classList.add('video-fullscreen');
      handleMouseActivity();
    } else {
      document.body.classList.remove('video-fullscreen');
      document.body.classList.remove('vs-hidden-ui');
    }
  };
  document.addEventListener('fullscreenchange', onFs);
  document.addEventListener('webkitfullscreenchange', onFs);
  document.addEventListener('mozfullscreenchange', onFs);
  document.addEventListener('MSFullscreenChange', onFs);
  document.addEventListener('mousemove', handleMouseActivity);
  document.addEventListener('touchstart', handleMouseActivity);
  document.addEventListener('click', handleMouseActivity);
}

function bindPlayerControls(){
  var pp = $('#vc-play-pause'); if (pp) pp.onclick = function(e){ e.stopPropagation(); togglePlayPause(); };
  var b10 = $('#vc-back10'); if (b10) b10.onclick = function(e){ e.stopPropagation(); seekBy(-10); };
  var f10 = $('#vc-fwd10'); if (f10) f10.onclick = function(e){ e.stopPropagation(); seekBy(10); };
  var spd = $('#vc-speed'); if (spd) spd.onchange = function(e){ setSpeed(parseFloat(spd.value)); };
  var dl = $('#vc-download'); if (dl) dl.onclick = function(e){ e.stopPropagation(); downloadCurrent(); };
  var fs = $('#vc-fullscreen'); if (fs) fs.onclick = function(e){ e.stopPropagation(); toggleFullscreen(); };
  var center = $('#vp-center-pause'); if (center) center.onclick = function(e){ e.stopPropagation(); togglePlayPause(); };
  var bar = $('#vp-progress-bar');
  if (bar) bar.onclick = function(e){
    e.stopPropagation();
    var rect = bar.getBoundingClientRect();
    var pct = (e.clientX - rect.left) / rect.width;
    if (PLAYER.duration > 0) seekTo(pct * PLAYER.duration);
  };
  var shield = $('#vp-shield');
  if (shield){
    shield.addEventListener('click', function(e){ e.stopPropagation(); handleBoxTap(e.clientX); });
    shield.addEventListener('touchend', function(e){ e.stopPropagation(); handleBoxTap(e.changedTouches[0].clientX); }, { passive: true });
  }
  bindFullscreenEvents();
  if (!window._atpMsgBound){
    window._atpMsgBound = true;
    window.addEventListener('message', function(e){
      var d = e.data;
      if (!d || typeof d !== 'object') return;
      if (d.type === 'atp-status'){
        PLAYER.isPlaying = !!d.playing;
        if (typeof d.duration === 'number' && d.duration > 0) PLAYER.duration = d.duration;
        if (typeof d.currentTime === 'number'){ PLAYER.currentTime = d.currentTime; PLAYER.lastPosition = d.currentTime; }
        updateTimeUI();
        updatePlayPauseUI();
      }
      if (d.type === 'atp-download-url' && d.url) PLAYER.progressUrl = d.url;
      if (d.type === 'atp-download-error') PLAYER.progressUrl = null;
    });
  }
}

function handleBoxTap(x){
  var now = Date.now();
  if (now - PLAYER.lastTapTime < 320){
    var box = $('#vp-box');
    var rect = box.getBoundingClientRect();
    var mid = rect.left + rect.width / 2;

// ============================================================
function startWatchTracking(){
  if (PLAYER.watchTimer) clearInterval(PLAYER.watchTimer);
  var lastPlayerTime = PLAYER.currentTime || 0;
  var lastSave = Date.now();

  PLAYER.watchTimer = setInterval(function(){
    var stage = $('#video-stage');
    if (!stage || stage.style.display === 'none') return;
    if (document.hidden) return; // Tab hidden, don't track
    
    var isPlaying = false;
    if (PLAYER.mode === 'youtube' && PLAYER.ytPlayer && PLAYER.ytPlayer.getPlayerState){
      try { isPlaying = PLAYER.ytPlayer.getPlayerState() === 1; } catch(e){}
    } else if (PLAYER.videoEl){
      isPlaying = !PLAYER.videoEl.paused;
    } else if (PLAYER.mode === 'streaming'){
      isPlaying = PLAYER.isPlaying;
    }
    
    if (!isPlaying) {
      lastPlayerTime = PLAYER.currentTime; // Sync time when paused
      return;
    }
    
    var currentTime = PLAYER.currentTime || 0;
    var delta = currentTime - lastPlayerTime;
    
    // If video moved forward naturally (not seeking forward)
    if (delta > 0 && delta < 5) {
      PLAYER.watchSeconds += delta;
      lastPlayerTime = currentTime;
    } else if (delta < 0) {
      lastPlayerTime = currentTime; // Video was rewound
    }

    var now = Date.now();
    if (now - lastSave > 10000 && PLAYER.lecture){
      lastSave = now;
      saveWatchProgress(PLAYER.lecture.id, PLAYER.watchSeconds, PLAYER.duration);
    }
  }, 1000);
}

      headers: {'Authorization': 'Bearer ' + STATE.session.access_token, 'Content-Type': 'application/json'},
      body: JSON.stringify({
        lectureId: lectureId,
        seconds: Math.floor(seconds),
        totalDuration: totalDuration || 0,
        completed: completed,
        lastPosition: Math.floor(PLAYER.lastPosition || 0),
        sourceType: 'video'
      })
    });
  } catch(e){}
}

function DownloadManager(){
  this.max = 2;
  this.chunksPerDownload = 4;
  this.chunkSize = 4 * 1024 * 1024;
  this.tasks = new Map();
  this.active = 0;
  this.queue = [];
}

DownloadManager.prototype.addDownload = function(id, name, type, url){
  if (this.tasks.has(id)){ toast('Already downloading'); return; }
  var task = {
    id: id, name: name, type: type, url: url,
    status: 'pending', progress: 0,
    downloaded: 0, total: 0,
    chunks: [], blob: null, savedAt: Date.now()
  };
  this.tasks.set(id, task);
  this.queue.push(id);
  this._process();
  this._updateBadge();
};

DownloadManager.prototype._process = async function(){
  if (this.active >= this.max) return;
  var id = this.queue.shift();
  if (!id) return;
  var task = this.tasks.get(id);
  if (!task || task.status !== 'pending'){ this._process(); return; }
  task.status = 'downloading';
  this.active++;
  this._updateBadge();
  var self = this;
  try {
    var r = await fetch('/api/proxy-download?url=' + encodeURIComponent(task.url));
    if (!r.ok) throw new Error('HTTP ' + r.status);
    var size = parseInt(r.headers.get('content-length') || '0');
    task.total = size;
    var reader = r.body.getReader();
    while (true){
      var res = await reader.read();
      if (res.done) break;
      task.chunks.push(res.value);
      task.downloaded += res.value.length;
      task.progress = size > 0 ? Math.min(99, Math.round((task.downloaded / size) * 100)) : Math.min(95, Math.round(task.downloaded / 1048576 * 2));
      if (task.downloaded % (1024 * 1024) < 65536) self._updateBadge();
    }
    task.blob = new Blob(task.chunks, { type: task.type === 'video' ? 'video/mp4' : 'application/pdf' });
    task.status = 'completed';
    task.progress = 100;
    await self._saveToIndexedDB(task);
  } catch(e){
    task.status = 'error';
    task.error = e.message;
  } finally {
    self.active--;
    self._process();
    self._updateBadge();
    var m2 = $('#downloads-modal');
    if (m2 && m2.style.display === 'flex') renderDownloads();
  }
};

DownloadManager.prototype._saveToIndexedDB = function(task){
  return new Promise(function(resolve){
    var req = indexedDB.open('vibhuux_offline', 1);
    req.onupgradeneeded = function(e){
      var db = e.target.result;
      if (!db.objectStoreNames.contains('files')) db.createObjectStore('files', { keyPath: 'id' });
    };
    req.onsuccess = function(e){
      var db = e.target.result;
      var tx = db.transaction('files', 'readwrite');
      tx.objectStore('files').put({
        id: task.id, name: task.name, type: task.type, blob: task.blob,
        url: task.url, savedAt: Date.now()
      });
      tx.oncomplete = function(){ resolve(); };
    };
    req.onerror = function(){ resolve(); };
  });
};

DownloadManager.prototype.getFromIndexedDB = function(id){
  return new Promise(function(resolve){
    var req = indexedDB.open('vibhuux_offline', 1);
    req.onsuccess = function(e){
      var db = e.target.result;
      if (!db.objectStoreNames.contains('files')) return resolve(null);
      var tx = db.transaction('files', 'readonly');
      var r = tx.objectStore('files').get(id);
      r.onsuccess = function(){ resolve(r.result); };
    };
    req.onerror = function(){ resolve(null); };
  });
};

DownloadManager.prototype.deleteFromIndexedDB = function(id){
  return new Promise(function(resolve){
    var req = indexedDB.open('vibhuux_offline', 1);
    req.onsuccess = function(e){
      var db = e.target.result;
      var tx = db.transaction('files', 'readwrite');
      tx.objectStore('files').delete(id);
      tx.oncomplete = function(){ resolve(); };
    };
  });
};

DownloadManager.prototype._updateBadge = function(){
  var count = 0;
  this.tasks.forEach(function(t){
    if (t.status === 'downloading' || t.status === 'pending') count++;
  });
  var badge = $('#download-badge');
  if (badge){
    if (count > 0){ badge.style.display = 'inline-block'; badge.textContent = count; }
    else badge.style.display = 'none';
  }
};

DownloadManager.prototype.getAll = function(){ return Array.from(this.tasks.values()); };
DownloadManager.prototype.pause = function(id){ var t = this.tasks.get(id); if (t) t.status = 'paused'; };
DownloadManager.prototype.resume = function(id){ var t = this.tasks.get(id); if (t){ t.status = 'pending'; this.queue.push(id); this._process(); } };
DownloadManager.prototype.cancel = function(id){ this.tasks.delete(id); this._updateBadge(); };
DownloadManager.prototype.retry = function(id){
  var t = this.tasks.get(id);
  if (t){ t.status = 'pending'; t.chunks = []; t.downloaded = 0; t.progress = 0; this.queue.push(id); this._process(); }
};

async function loadExistingDownloads(){
  return new Promise(function(resolve){
    try {
      var req = indexedDB.open('vibhuux_offline', 1);
      req.onupgradeneeded = function(e){
        var db = e.target.result;
        if (!db.objectStoreNames.contains('files')) db.createObjectStore('files', { keyPath: 'id' });
      };
      req.onsuccess = function(e){
        var db = e.target.result;
        if (!db.objectStoreNames.contains('files')) return resolve();
        var tx = db.transaction('files', 'readonly');
        var all = tx.objectStore('files').getAll();
        all.onsuccess = function(){
          (all.result || []).forEach(function(file){
            if (!downloadManager.tasks.has(file.id)){
              downloadManager.tasks.set(file.id, {
                id: file.id, name: file.name, type: file.type, url: file.url,
                status: 'completed', progress: 100, blob: file.blob,
                downloaded: file.blob ? file.blob.size : 0,
                total: file.blob ? file.blob.size : 0,
                chunks: [], savedAt: file.savedAt || 0
              });
            }
          });
          downloadManager._updateBadge();
          resolve();
        };
      };
      req.onerror = function(){ resolve(); };
    } catch(e){ resolve(); }
  });
}

window._openChat = async function(chatId, name, username){

  $('#chat-title').textContent = name + ' (@' + username + ')';
  $('#chat-modal').style.display = 'flex';
  $('#chat-typing-indicator').style.display = 'none';
  $('#chat-media-preview').style.display = 'none';
  await loadMessages(chatId);
  await markChatSeen(chatId);
  await loadMediaLimit();
  if (STATE.chatPollTimer) clearInterval(STATE.chatPollTimer);
  STATE.chatPollTimer = setInterval(function(){ loadMessages(chatId); }, 3000);
  if (STATE.chatOtherTypingTimer) clearInterval(STATE.chatOtherTypingTimer);
  STATE.chatOtherTypingTimer = setInterval(function(){ checkOtherTyping(chatId); }, 2000);
};

async function markChatSeen(chatId){
  try {
    await fetch('/api/chat/mark-seen', {
      method: 'POST',
      headers: {'Authorization': 'Bearer ' + STATE.session.access_token, 'Content-Type': 'application/json'},
      body: JSON.stringify({ chatId: chatId })
    });
    updateChatBadge();
  } catch(e){}
}

async function loadMessages(chatId){
  try {
    var r = await fetch('/api/chat/messages?chat_id=' + chatId, { headers: {'Authorization': 'Bearer ' + STATE.session.access_token} });
    var data = await r.json();
    var msgs = data.messages || [];
    var el = $('#chat-messages');
    if (!el) return;
    var html = msgs.map(function(m){
      var self = m.sender_id === STATE.user.id;
      var t = new Date(m.timestamp).toLocaleTimeString([], { hour:'2-digit', minute:'2-digit' });
      var tickHtml = '';
      if (self){
        var tickClass = 'chat-ticks';
        if (m.seen_at) tickClass += ' seen';
        if (m.delivered_at || m.seen_at){
          tickHtml = '<span class="' + tickClass + '"><svg viewBox="0 0 18 12"><path class="tick-path" d="M1 6 L4 9 L10 2" /><path class="tick-path" d="M6 6 L9 9 L15 2" /></svg></span>';
        } else {
          tickHtml = '<span class="' + tickClass + '"><svg viewBox="0 0 18 12"><path class="tick-path" d="M1 6 L4 9 L10 2" /></svg></span>';
        }
      }
      var body = '';
      if (m.message_type === 'image' && m.media_url){
        body = '<div class="chat-bubble img-bubble"><img src="' + escapeAttr(m.media_url) + '" onclick="window._openImageViewer(\'' + escapeJs(m.media_url) + '\')" loading="lazy"></div>';
      } else {
        body = '<div class="chat-bubble">' + escapeHtml(m.text || '') + '</div>';
      }
      return '<div class="chat-msg ' + (self ? 'self' : 'other') + '">' + body + '<div class="chat-meta">' + t + ' ' + tickHtml + '</div></div>';
    }).join('');
    el.innerHTML = html || '<div class="empty-state" style="padding:20px;"><p>Say hi! 👋</p></div>';
    el.scrollTop = el.scrollHeight;
  } catch(e){}
}

async function sendChatMessage(){
  var text = $('#chat-input').value.trim();
  if (!text || !STATE.currentChat) return;
  $('#chat-input').value = '';
  try {
    await fetch('/api/chat/send', {
      method: 'POST',
      headers: {'Authorization': 'Bearer ' + STATE.session.access_token, 'Content-Type': 'application/json'},
      body: JSON.stringify({ chatId: STATE.currentChat.chatId, text: text, type: 'text' })
    });
    await loadMessages(STATE.currentChat.chatId);
    stopTyping();
  } catch(e){ toast(e.message, 'error'); }
}

var typingSentTime = 0;
async function sendTyping(){
  if (!STATE.currentChat) return;
  var now = Date.now();
  if (now - typingSentTime < 1500) return;
  typingSentTime = now;
  try {
    await fetch('/api/chat/typing', {
      method: 'POST',
      headers: {'Authorization': 'Bearer ' + STATE.session.access_token, 'Content-Type': 'application/json'},
      body: JSON.stringify({ chatId: STATE.currentChat.chatId })
    });
  } catch(e){}
}
function stopTyping(){ typingSentTime = 0; }

async function checkOtherTyping(chatId){
  try {
    var r = await fetch('/api/chat/typing-status?chat_id=' + chatId, { headers: {'Authorization': 'Bearer ' + STATE.session.access_token} });
    var data = await r.json();
    var el = $('#chat-typing-indicator');
    if (data.typing){
      el.textContent = data.name + ' typing...';
      el.style.display = 'inline-block';
    } else {
      el.style.display = 'none';
    }
  } catch(e){}
}

async function loadMediaLimit(){
  try {
    var r = await fetch('/api/chat/media-limit', { headers: {'Authorization': 'Bearer ' + STATE.session.access_token} });
    var data = await r.json();
    STATE.mediaUploadsToday = data.used || 0;
    STATE.mediaUploadLimit = data.limit || 0;
    updateMediaBtn();
  } catch(e){}
}
function updateMediaBtn(){
  var btn = $('#chat-media-btn');
  var badge = $('#chat-media-limit-badge');
  if (!btn || !badge) return;
  if (STATE.mediaUploadLimit === 0){
    btn.disabled = false;
    badge.style.display = 'none';
    btn.title = 'Send image (unlimited)';
  } else {
    var remaining = STATE.mediaUploadLimit - STATE.mediaUploadsToday;
    btn.disabled = remaining <= 0;
    if (remaining > 0){
      badge.style.display = 'none';
      btn.title = 'Send image (' + remaining + ' remaining today)';
    } else {
      badge.style.display = 'inline-block';
      badge.textContent = '0';
      btn.title = 'Daily limit reached';
    }
  }
}

function setupChatMedia(){
  var mediaBtn = $('#chat-media-btn');
  var fileInput = $('#chat-file-input');
  if (!mediaBtn || !fileInput) return;
  mediaBtn.onclick = function(){
    if (mediaBtn.disabled) return toast('Daily limit reached', 'error');
    fileInput.click();
  };
  fileInput.onchange = function(e){
    var file = e.target.files[0];
    if (!file) return;
    if (!file.type.startsWith('image/')) return toast('Only images allowed', 'error');
    if (file.size > 5 * 1024 * 1024) return toast('Max 5MB', 'error');
    showMediaPreview(file);
    fileInput.value = '';
  };
  var sendBtn = $('#chat-send-btn');
  if (sendBtn) sendBtn.onclick = sendChatMessage;
  var input = $('#chat-input');
  if (input) input.addEventListener('keypress', function(e){
    if (e.key === 'Enter'){ e.preventDefault(); sendChatMessage(); }
  });
  if (input) input.addEventListener('input', function(){ sendTyping(); });
}

var pendingMediaFile = null;
var pendingMediaUrl = null;

function showMediaPreview(file){
  pendingMediaFile = file;
  if (pendingMediaUrl) URL.revokeObjectURL(pendingMediaUrl);
  pendingMediaUrl = URL.createObjectURL(file);
  var preview = $('#chat-media-preview');
  preview.innerHTML = '<img src="' + pendingMediaUrl + '"><div class="cmp-info"><strong>' + escapeHtml(file.name) + '</strong><br>' + (file.size/1024).toFixed(0) + ' KB</div><div class="cmp-actions"><button onclick="window._cancelMedia()">Cancel</button><button style="background:linear-gradient(135deg,#22c55e,#16a34a);color:#fff;border:none;" onclick="window._sendMedia()">Send</button></div>';
  preview.style.display = 'flex';
}

window._cancelMedia = function(){
  pendingMediaFile = null;
  if (pendingMediaUrl){ URL.revokeObjectURL(pendingMediaUrl); pendingMediaUrl = null; }
  $('#chat-media-preview').style.display = 'none';
};

window._sendMedia = async function(){
  if (!pendingMediaFile || !STATE.currentChat) return;
  var btn = event && event.target ? event.target : null;
  if (btn){ btn.disabled = true; btn.textContent = 'Uploading...'; }
  try {
    var fd = new FormData();
    fd.append('file', pendingMediaFile);
    var r = await fetch('/api/chat/upload-media', {
      method: 'POST',
      headers: {'Authorization': 'Bearer ' + STATE.session.access_token},
      body: fd
    });
    var data = await r.json();
    if (!r.ok) throw new Error(data.error || 'Upload failed');
    await fetch('/api/chat/send', {
      method: 'POST',
      headers: {'Authorization': 'Bearer ' + STATE.session.access_token, 'Content-Type': 'application/json'},
      body: JSON.stringify({
        chatId: STATE.currentChat.chatId,
        text: '',
        type: 'image',
        mediaUrl: data.url,
        mediaSize: pendingMediaFile.size
      })
    });
    window._cancelMedia();
    await loadMessages(STATE.currentChat.chatId);
    await loadMediaLimit();
  } catch(e){ toast('❌ ' + e.message, 'error'); }
  finally { if (btn){ btn.disabled = false; btn.textContent = 'Send'; } }
};

window._submitPaidRequest = async function(){
  var errEl = $('#pr-error'); if (errEl) errEl.textContent = '';
  var btn = $('#pr-submit-btn');
  var aspirant = $('#pr-aspirant').value.trim();
  var paymentName = $('#pr-payment').value.trim();
  var utr = $('#pr-utr').value.trim();
  var paymentDate = $('#pr-date').value;
  var batches = $$('#pr-batches input:checked').map(function(c){ return c.value; });
  var files = Array.prototype.slice.call($('#pr-files').files).slice(0,3);

  if (!aspirant){ if (errEl) errEl.textContent = 'Aspirant name required'; return; }
  if (!paymentName){ if (errEl) errEl.textContent = 'Payment name required'; return; }
  if (!utr){ if (errEl) errEl.textContent = 'UTR required'; return; }
  if (!paymentDate){ if (errEl) errEl.textContent = 'Payment date required'; return; }
  if (!batches.length){ if (errEl) errEl.textContent = 'Kam se kam ek batch select kare'; return; }
  if (!files.length){ if (errEl) errEl.textContent = 'Payment screenshot upload kare'; return; }

  if (btn){ btn.disabled = true; btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Uploading...'; }
  try {
    var screenshotUrls = [];
    for (var i = 0; i < files.length; i++){
      var fd = new FormData();
      fd.append('file', files[i]);
      var r = await fetch('/api/paid/upload-screenshot', {
        method: 'POST',
        headers: {'Authorization': 'Bearer ' + STATE.session.access_token},
        body: fd
      });
      var txt = await r.text();
      if (!r.ok) {
        if (txt.includes('Auth required') || r.status === 401) {
          throw new Error('Session expired. Please login again.');
        }
        throw new Error('Upload fail: ' + txt.slice(0, 150));
      }
      var d = JSON.parse(txt);
      if (d.url) screenshotUrls.push(d.url);
      else throw new Error(d.error || 'URL not returned');
    }
    var sr = await fetch('/api/paid/submit', {
      method: 'POST',
      headers: {'Authorization': 'Bearer ' + STATE.session.access_token, 'Content-Type': 'application/json'},
      body: JSON.stringify({
        aspirantName: aspirant, paymentName: paymentName, utrNumber: utr,
        paymentDate: paymentDate, batches: batches, screenshotUrls: screenshotUrls
      })
    });
    var sd = await sr.json();
    if (!sr.ok || !sd.ok) throw new Error(sd.error || 'Submit fail');
    toast('✅ Request submitted successfully!');
    setTimeout(showPaidForm, 1200);
  } catch(e){
    if (errEl) errEl.textContent = e.message || 'Submit fail';
    toast('❌ ' + (e.message || 'Fail'), 'error');
    if (e.message.includes('Session expired') || e.message.includes('Auth')) {
      setTimeout(() => { doLogout(); }, 2000);
    }
  } finally {
    if (btn){ btn.disabled = false; btn.innerHTML = '<i class="fas fa-paper-plane"></i> Submit Request'; }
  }
};

      if (mlData.status === 'expired' || mlData.status === 'not_found') break;
      await new Promise(function(r){ setTimeout(r, 2000); });
    }
    if (verifiedEmail) {
      var mlCr = await fetch('/api/auth/tg-create', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ code: urlSetupCode, email: verifiedEmail, deviceId: STATE.deviceId, fingerprint: STATE.fingerprint })
      });
      var mlCd = await mlCr.json();
      if (!mlCr.ok) throw new Error(mlCd.error || 'Account setup failed');
      STATE.session = mlCd.session;
      STATE.user = mlCd.user;
      localStorage.setItem('vibhuux_session', JSON.stringify(mlCd.session));
      showProfileSetup();
      return;
    } else {
      toast('Magic link expired. Please signup again.', 'error');
    }
  } catch(e) {
    toast(e.message || 'Magic link failed', 'error');
  }
}
